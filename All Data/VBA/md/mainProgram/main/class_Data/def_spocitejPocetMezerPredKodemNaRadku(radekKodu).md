def spocitejPocetMezerPredKodemNaRadku(radekKodu):
    pocetMezer = 0

    znakyRadku = list(radekKodu)
    for znak in znakyRadku:
        if(znak.isspace() == True):
            pocetMezer = pocetMezer + 1

    return(pocetMezer)
